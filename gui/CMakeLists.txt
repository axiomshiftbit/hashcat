cmake_minimum_required(VERSION 3.21)
project(HashcatDesktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Concurrent)
add_library(hashcat_commandbuilder
  third_party/hashcat_gui/src/commandbuilder.cpp
  third_party/hashcat_gui/src/hashcatinfoparser.cpp
  third_party/hashcat_gui/src/helperutils.cpp
  third_party/hashcat_gui/src/settingsmanager.cpp)
target_include_directories(hashcat_commandbuilder PUBLIC third_party/hashcat_gui/src)
target_link_libraries(hashcat_commandbuilder PUBLIC Qt6::Widgets Qt6::Concurrent)
add_library(desktop_core src/job.cpp src/runner.cpp src/converter.cpp src/extractor.cpp)
qt_add_resources(desktop_core "extractor_registry" PREFIX "/" FILES third_party/hashcat_reactor/tool_registry.json)
target_include_directories(desktop_core PUBLIC src)
target_link_libraries(desktop_core PUBLIC Qt6::Widgets hashcat_commandbuilder)
add_library(desktop_ui src/window.cpp src/pathlist.cpp src/advancedform.cpp src/telemetry.cpp)
target_link_libraries(desktop_ui PUBLIC desktop_core)
add_executable(hashcat-desktop WIN32 src/main.cpp)
target_link_libraries(hashcat-desktop PRIVATE desktop_ui)
if(MSVC)
  target_compile_options(desktop_core PRIVATE /W4 /utf-8)
  target_compile_options(desktop_ui PRIVATE /W4 /utf-8)
else()
  target_compile_options(desktop_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(desktop_ui PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_link_libraries(desktop_core PRIVATE user32)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(desktop_tests tests/test_desktop.cpp)
  target_link_libraries(desktop_tests PRIVATE desktop_ui Qt6::Test)
  add_executable(fake_hashcat tests/fake_hashcat.cpp)
  target_link_libraries(fake_hashcat PRIVATE Qt6::Core)
  add_test(NAME desktop COMMAND desktop_tests)
  foreach(suite commandbuilder hashcatinfoparser helperutils)
    add_executable(upstream_${suite} third_party/hashcat_gui/tests/tst_${suite}.cpp)
    target_link_libraries(upstream_${suite} PRIVATE hashcat_commandbuilder Qt6::Test)
    add_test(NAME upstream_${suite} COMMAND upstream_${suite})
    set_tests_properties(upstream_${suite} PROPERTIES TIMEOUT 60)
  endforeach()
  set_tests_properties(desktop PROPERTIES TIMEOUT 300)
  if(NOT WIN32)
    set_tests_properties(desktop PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endif()
install(TARGETS hashcat-desktop RUNTIME DESTINATION bin)
